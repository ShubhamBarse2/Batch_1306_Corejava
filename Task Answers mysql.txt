Customer -> CustomerId, CusName ,  email, mobNo , address , country , state , paymentMode , status , orderId , ProductId


Performed Joins Operation : 


ref queries : 


CREATE TABLE Orders (
    orderId INT PRIMARY KEY,
    orderName VARCHAR(50),
    date DATE,
    status VARCHAR(20)
);

CREATE TABLE Product (
    productId INT PRIMARY KEY,
    productName VARCHAR(50),
    price DECIMAL(10,2),
    quantity INT,
    manfdate DATE,
    type VARCHAR(30)
);

CREATE TABLE Customer (
    customerId INT PRIMARY KEY,
    cusName VARCHAR(50),
    email VARCHAR(50),
    mobNo VARCHAR(15),
    address VARCHAR(100),
    country VARCHAR(30),
    state VARCHAR(30),
    paymentMode VARCHAR(20),
    status VARCHAR(20),
    orderId INT,
    productId INT,
    FOREIGN KEY (orderId) REFERENCES Orders(orderId),
    FOREIGN KEY (productId) REFERENCES Product(productId)
);

1. Inner join : 

SELECT c.cusName, o.orderName, p.productName, p.price
FROM Customer c
INNER JOIN Orders o ON c.orderId = o.orderId
INNER JOIN Product p ON c.productId = p.productId;




2. LEFT JOIN
 All customers + matching orders/products

SELECT c.cusName, o.orderName, p.productName
FROM Customer c
LEFT JOIN Orders o ON c.orderId = o.orderId
LEFT JOIN Product p ON c.productId = p.productId;




3. RIGHT JOIN
All orders + matching customers

SELECT c.cusName, o.orderName
FROM Customer c
RIGHT JOIN Orders o ON c.orderId = o.orderId;



4. FULL JOIN 
 All records from both sides

SELECT c.cusName, o.orderName
FROM Customer c
LEFT JOIN Orders o ON c.orderId = o.orderId
UNION
SELECT c.cusName, o.orderName
FROM Customer c
RIGHT JOIN Orders o ON c.orderId = o.orderId;



5. JOIN with WHERE Condition -> 

SELECT c.cusName, p.productName, p.price
FROM Customer c
INNER JOIN Product p ON c.productId = p.productId
WHERE p.price > 30000;


6. JOIN with GROUP BY -> 

SELECT p.type, COUNT(*) AS total_products
FROM Product p
INNER JOIN Customer c ON p.productId = c.productId
GROUP BY p.type;


7. JOIN with ORDER BY -> 

SELECT c.cusName, p.productName, p.price
FROM Customer c
INNER JOIN Product p ON c.productId = p.productId
ORDER BY p.price DESC;
